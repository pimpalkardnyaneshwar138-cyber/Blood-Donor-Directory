# ================= ANALYZE =================

def analyze():

    student = [
        level_score[skill_vars[s].get()]
        for s in skills
    ]

    selected_interest = interest.get()

    results = []

    for career, required in careers.items():

        score = sum(
            100 - abs(a - b)
            for a, b in zip(student, required)
        ) / len(skills)

        # Interest bonus
        bonus = 0

        if selected_interest == "Artificial Intelligence / ML" and \
                career in ["Machine Learning Engineer", "Data Scientist"]:
            bonus = 8

        elif selected_interest == "Data Science" and \
                career in ["Data Scientist", "Data Analyst"]:
            bonus = 8

        elif selected_interest == "Software Development" and \
                career == "Software Developer":
            bonus = 8

        elif selected_interest == "Web Development" and \
                career == "Web Developer":
            bonus = 8

        elif selected_interest == "Cyber Security" and \
                career == "Cyber Security Analyst":
            bonus = 8

        elif selected_interest == "Cloud Computing" and \
                career == "Cloud Engineer":
            bonus = 8

        final = min(
            100,
            round(score * .90 + bonus, 2)
        )

        results.append((career, final))

    results.sort(
        key=lambda x: x[1],
        reverse=True
    )

    # Best career
    career_label.config(
        text=results[0][0]
    )

    score_label.config(
        text=f"{results[0][1]}%"
    )

    # Top 5
    text = ""

    for i, (career, score) in enumerate(
        results[:5], 1
    ):
        text += f"{i}.  {career}\n"
        text += f"     Match Score: {score}%\n\n"

    ranking.config(text=text)

    # Strengths
    strengths = []

    for skill, value in zip(skills, student):

        if value >= 75:
            strengths.append(
                f"✓ Strong {skill}"
            )

    if not strengths:
        strengths.append(
            "✓ Keep improving your technical skills"
        )

    reason.config(
        text="\n".join(strengths[:4]),
        fg=WHITE
    )
